# ============================================================================
# INSTALLATION NOTES FOR LINUX SERVER
# ============================================================================
#
# 1. Create virtual environment:
#    python3 -m venv venv
#    source venv/bin/activate
#
# 2. Upgrade pip:
#    pip install --upgrade pip setuptools wheel
#
# 3. Install requirements:
#    pip install -r requirements.txt
#
# 4. For production with gunicorn:
#    gunicorn fastapi_chatbot:app -w 4 -k uvicorn.workers.UvicornWorker
#
# 5. For Redis (if not installed):
#    sudo apt-get install redis-server
#    sudo systemctl start redis
#
# 6. For Ollama (if not installed):
#    curl https://ollama.ai/install.sh | sh
#    ollama pull deepseek-v3.1:671b-cloud
#
# ============================================================================
# LINUX-SPECIFIC CONSIDERATIONS
# ============================================================================
#
# - faiss-cpu 1.8.0 is the most stable for Linux
# - hiredis provides 5-10x faster Redis performance on Linux
# - Use systemd for process management in production
# - Consider nginx as reverse proxy
# - Set up logrotate for log management
